   407→function umsg(e){
   408→  const t=e&&e.message?e.message:String(e);
   409→  // On-chain TVM errors (from gosh.runwasm verifier)
   410→  if(/wasm failed to load/i.test(t))return'On-chain verifier error — app_id not registered or mining keys not activated';
   411→  if(/wasm failed to execute/i.test(t))return'On-chain execution failed — tap verification rejected';
   412→  if(/wasm forbidden binary/i.test(t))return'Verifier binary forbidden — contact support';
   413→  if(/wasm invalid function/i.test(t))return'Verifier function not found — SDK version mismatch';
   414→  if(/wasm linker failed/i.test(t))return'Verifier linker error — SDK version mismatch';
   415→  if(/wasm failed to instantiate/i.test(t))return'Verifier instantiation failed — chain issue';
   416→  if(/wasm engine not created/i.test(t))return'Chain wasm engine unavailable — try later';
   417→  if(/wasm hash not in whitelist/i.test(t))return'Verifier hash not whitelisted — SDK outdated';
   418→  if(/DApp ID is not set|DAppIdNotSet/i.test(t))return'DApp ID not set — waiting for your registered app_id';
   419→  if(/wasm config/i.test(t))return'Chain wasm config error — contact support';
   420→  if(/not ready|ERR_NOT_READY/i.test(t))return'Mining not ready — activate Mambaboard in AN Wallet first';
   421→  // Network errors
   422→  if(/pool timed out|open connection/i.test(t))return'Network timeout — retrying';
   423→  if(/memory access out of bounds|RuntimeError/i.test(t))return'WASM memory error';
   424→  if(/FnOnce called more than once/i.test(t))return'WASM callback error';
   425→  if(/Invalid ABI|Decode message body/i.test(t))return'ABI decode error';
   426→  return t.slice(0,200)
